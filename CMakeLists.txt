cmake_minimum_required(VERSION 3.16)
project(apxgen CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W3 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra)
endif()
add_library(apxgen STATIC src/mission.cpp src/program.cpp src/layout.cpp src/critic.cpp src/dress.cpp src/vox.cpp)
target_include_directories(apxgen PUBLIC include)
add_executable(apxgen_cli tools/apxgen_cli.cpp)
target_link_libraries(apxgen_cli PRIVATE apxgen)
target_compile_definitions(apxgen_cli PRIVATE APXGEN_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(apxgen_cli PROPERTIES OUTPUT_NAME apxgen)

enable_testing()
add_test(NAME w2_gate COMMAND apxgen_cli --seeds 1000 --quiet --max-retried 10)
add_test(NAME w2_control_corrupt_grammar COMMAND apxgen_cli --seeds 50 --quiet --grammar ${CMAKE_CURRENT_SOURCE_DIR}/grammar/corrupt-key-behind-lock.grammar)
set_tests_properties(w2_control_corrupt_grammar PROPERTIES WILL_FAIL TRUE)
add_test(NAME dig_control_no_hard_walls COMMAND apxgen_cli --seeds 50 --quiet --no-hard-walls)
set_tests_properties(dig_control_no_hard_walls PROPERTIES WILL_FAIL TRUE)
# F3: .vox write + read-back, dress with Clone Army as a gate, and its control (cap off must fail)
add_test(NAME f3_vox_roundtrip COMMAND apxgen_cli --vox ${CMAKE_CURRENT_BINARY_DIR}/vox)
add_test(NAME f3_dress_strict COMMAND apxgen_cli --seeds 200 --quiet --strict-soft)
add_test(NAME f3_control_no_clone_cap COMMAND apxgen_cli --seeds 50 --quiet --strict-soft --no-clone-cap)
set_tests_properties(f3_control_no_clone_cap PROPERTIES WILL_FAIL TRUE)
# all five special rooms on every floor; control: special nodes keep their BFS cell (must fail)
add_test(NAME f3_control_no_special_cells COMMAND apxgen_cli --seeds 50 --quiet --no-special-cells)
set_tests_properties(f3_control_no_special_cells PROPERTIES WILL_FAIL TRUE)
# retries: under 10 per 1000 floors; each mechanism off brings the dead ends back (must fail)
add_test(NAME f3_control_no_edge_reorder COMMAND apxgen_cli --seeds 1000 --quiet --no-dress --max-retried 10 --no-edge-reorder)
set_tests_properties(f3_control_no_edge_reorder PROPERTIES WILL_FAIL TRUE)
add_test(NAME f3_control_no_band_steal COMMAND apxgen_cli --seeds 1000 --quiet --no-dress --max-retried 10 --no-band-steal)
set_tests_properties(f3_control_no_band_steal PROPERTIES WILL_FAIL TRUE)
